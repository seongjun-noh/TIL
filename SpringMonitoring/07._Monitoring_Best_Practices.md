## 1. 메트릭 수집 전략

### 핵심 메트릭 정의
```yaml
core_metrics:
  system:
    - name: CPU 사용률
      type: gauge
      importance: critical
    - name: 메모리 사용률
      type: gauge
      importance: critical
    - name: 디스크 사용률
      type: gauge
      importance: warning
      
  application:
    - name: 응답 시간
      type: histogram
      importance: critical
    - name: 에러율
      type: gauge
      importance: critical
    - name: 처리량
      type: counter
      importance: warning
      
  business:
    - name: 활성 사용자 수
      type: gauge
      importance: info
    - name: 트랜잭션 성공률
      type: gauge
      importance: critical
    - name: 매출 지표
      type: counter
      importance: warning
```

### 커스텀 메트릭 구현
```java
@Configuration
public class MetricsConfig {
    
    @Bean
    public MeterRegistry meterRegistry() {
        CompositeMeterRegistry registry = new CompositeMeterRegistry();
        
        // 비즈니스 메트릭
        Counter orderCounter = Counter.builder("business.orders")
            .tag("type", "total")
            .description("총 주문 수")
            .register(registry);
            
        Gauge userGauge = Gauge.builder("business.active.users", 
            userService, this::getActiveUserCount)
            .description("활성 사용자 수")
            .register(registry);
            
        // 성능 메트릭
        Timer requestTimer = Timer.builder("app.request.duration")
            .tag("endpoint", "/api/v1")
            .description("API 요청 처리 시간")
            .register(registry);
            
        return registry;
    }
}
```

### 태깅 전략
```java
@Component
public class MetricsTagger {
    
    private final MeterRegistry registry;
    
    public void tagRequest(HttpServletRequest request) {
        Tags tags = Tags.of(
            Tag.of("endpoint", request.getRequestURI()),
            Tag.of("method", request.getMethod()),
            Tag.of("status", response.getStatus()),
            Tag.of("client", request.getHeader("User-Agent")),
            Tag.of("region", getClientRegion(request))
        );
        
        // 메트릭에 태그 적용
        registry.counter("http.requests", tags).increment();
    }
}
```

## 3. 보안 설정

### Actuator 엔드포인트 보안
```yaml
# application.yml
management:
  endpoints:
    web:
      exposure:
        include: health,info,metrics
      base-path: /management
      cors:
        allowed-origins: "https://monitoring.example.com"
        allowed-methods: GET
  endpoint:
    health:
      show-details: when_authorized
    shutdown:
      enabled: false

spring:
  security:
    user:
      name: ${ACTUATOR_USER}
      password: ${ACTUATOR_PASSWORD}
      roles: ACTUATOR_ADMIN
```

### 인증/인가 설정
```java
@Configuration
@EnableWebSecurity
public class ActuatorSecurityConfig extends WebSecurityConfigurerAdapter {
    
    @Override
    protected void configure(HttpSecurity http) throws Exception {
        http
            .requestMatcher(EndpointRequest.toAnyEndpoint())
            .authorizeRequests()
                .requestMatchers(EndpointRequest.to("health", "info"))
                    .permitAll()
                .requestMatchers(EndpointRequest.toAnyEndpoint())
                    .hasRole("ACTUATOR_ADMIN")
            .and()
            .httpBasic();
    }
}
```

### 민감 정보 보호
```java
@Configuration
public class MetricsSecurityConfig {
    
    @Bean
    public MeterFilter meterFilter() {
        return MeterFilter.deny(id -> {
            String name = id.getName();
            return name.startsWith("secret") || 
                   name.contains("password") ||
                   name.contains("key");
        });
    }
    
    @Bean
    public HttpTraceRepository httpTraceRepository() {
        return new InMemoryHttpTraceRepository() {
            @Override
            public List<HttpTrace> findAll() {
                return super.findAll().stream()
                    .map(this::sanitizeTrace)
                    .collect(Collectors.toList());
            }
            
            private HttpTrace sanitizeTrace(HttpTrace trace) {
                // 민감 정보 제거
                trace.getRequest().getHeaders().remove("Authorization");
                trace.getRequest().getHeaders().remove("Cookie");
                return trace;
            }
        };
    }
}
```

## 4. 모니터링 운영 가이드

### 일일 점검 항목
```yaml
daily_checks:
  system:
    - CPU 사용률 추이 확인
    - 메모리 누수 징후 확인
    - 디스크 공간 확인
    
  application:
    - 에러 로그 검토
    - 성능 지표 확인
    - 활성 사용자 수 확인
    
  security:
    - 비정상 접근 시도 확인
    - 인증 실패 로그 검토
    - 권한 변경 이력 확인
```

### 주간 리뷰 항목
```yaml
weekly_reviews:
  performance:
    - 응답 시간 추이 분석
    - 리소스 사용량 패턴 분석
    - 병목 구간 식별
    
  capacity:
    - 리소스 확장 필요성 검토
    - 부하 테스트 결과 분석
    - 스케일링 계획 수립
    
  maintenance:
    - 알림 임계값 조정
    - 대시보드 최적화
    - 메트릭 보존 기간 조정
```

### 월간 개선 계획
```yaml
monthly_improvements:
  metrics:
    - 불필요한 메트릭 제거
    - 새로운 비즈니스 메트릭 추가
    - 메트릭 수집 주기 최적화
    
  dashboards:
    - 대시보드 사용성 개선
    - 새로운 시각화 추가
    - 리포트 자동화 구현
    
  automation:
    - 반복 작업 자동화
    - 알림 규칙 개선
    - 장애 대응 자동화
```
